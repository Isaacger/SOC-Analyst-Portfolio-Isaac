# Run in ADMIN PowerShell: reading the Sysmon log needs admin rights.
# Note: the Where-Object filter hides events without the marker (e.g. whoami.exe, Event 3), so check without it too.
$start = (Get-Date).AddMinutes(-45)
Get-WinEvent -FilterHashtable @{LogName='Microsoft-Windows-Sysmon/Operational'; Id=1,3,11,22; StartTime=$start} | ForEach-Object {
 $d=@{}; ([xml]$_.ToXml()).Event.EventData.Data | ForEach-Object { $d[$_.Name]=$_.'#text' }
 [pscustomobject]@{UTC=$_.TimeCreated.ToUniversalTime().ToString('HH:mm:ss'); Id=$_.Id; Image=$d.Image; Cmd=$d.CommandLine
  Guid=$d.ProcessGuid; ParentGuid=$d.ParentProcessGuid; Dest=$d.DestinationHostname; Query=$d.QueryName; File=$d.TargetFilename; Integrity=$d.IntegrityLevel}
} | Where-Object { "$($_.Cmd)$($_.Query)$($_.File)$($_.Dest)" -match 'p03|example' } | Format-List
